/** * New CI Protection Planner 雲端中央後端 (v9.6.9 升級版) * 支援秒開版本檢查 (checkVersion)、尊嚴價值頁面 (highlightText / imageUrl)、條款 displayNote 與雙向同步 */ function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 系統設定 (Config) const configSheet = ss.getSheetByName("系統設定"); let config = {}; if (configSheet) { const cData = configSheet.getDataRange().getValues(); for (let i = 1; i < cData.length; i++) { if (cData[i][0]) config[cData[i][0]] = cData[i][2]; } } // ⚡ 輕量版本驗證 (支援秒開比對) const action = e && e.parameter ? e.parameter.action : ""; const callback = e && e.parameter ? e.parameter.callback : null; const currentVer = config["資料版本"] || "9.6.9"; if (action === "checkVersion") { const verPayload = { status: "success", version: currentVer, config: { "資料版本": currentVer } }; const verJson = JSON.stringify(verPayload); if (callback) { return ContentService.createTextOutput(`${callback}(${verJson})`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(verJson) .setMimeType(ContentService.MimeType.JSON); } // 2. 頁面文案 (Pages - 支援 highlightText 與 imageUrl) const pagesSheet = ss.getSheetByName("頁面文案"); let pages = []; if (pagesSheet) { const pData = pagesSheet.getDataRange().getValues(); for (let i = 1; i < pData.length; i++) { const row = pData[i]; if (!row[0] && !row[1]) continue; const roleText = String(row[1] || '').trim(); const role = (roleText === '兒童' || roleText === 'child') ? 'child' : 'adult'; const stepNum = Number(row[0]) || i; pages.push({ stepOrder: stepNum, role: role, pageId: String(row[5] || ("step_ci_" + stepNum)), pageType: String(row[6] || "文案"), tagName: String(row[2] || ''), title: String(row[3] || ''), subtitle: String(row[4] || ''), highlightText: String(row[7] || ''), imageUrl: String(row[8] || '') }); } } // 3. 選項設定 (Options) const optionsSheet = ss.getSheetByName("選項設定"); let options = []; if (optionsSheet) { const oData = optionsSheet.getDataRange().getValues(); for (let i = 1; i < oData.length; i++) { const row = oData[i]; if (!row[0] && !row[2]) continue; const pageBelong = String(row[0] || '').trim(); const roleText = String(row[1] || '').trim(); let role = 'common'; if (roleText === '成人' || roleText === 'adult') role = 'adult'; else if (roleText === '兒童' || roleText === 'child') role = 'child'; const defText = String(row[5] || '').trim(); const isDef = (defText === '是' || defText.toUpperCase() === 'TRUE') ? 'TRUE' : 'FALSE'; options.push({ pageId: String(row[7] || (pageBelong.indexOf('01') !== -1 ? 'step_ci_1' : 'step_ci_2')), role: role, optionKey: String(row[2] || ''), icon: String(row[3] || ''), title: String(row[4] || ''), desc: '', isDefaultSelected: isDef, order: Number(row[6]) || i }); } } // 4. 賠償案例 (Cases) const casesSheet = ss.getSheetByName("賠償案例"); let cases = casesSheet ? sheetToObjects(casesSheet) : []; // 5. 賠償條款 (Rules) const rulesSheet = ss.getSheetByName("賠償條款"); let rules = rulesSheet ? parseRulesSheet(rulesSheet) : []; // 6. 保費表 (每 HK$7,800 基準) const rates = { oys2_m: parseOys2RateTable(ss.getSheetByName("男愛伴航2保費")), oys2_f: parseOys2RateTable(ss.getSheetByName("女愛伴航2保費")), sce_m: parseSceRateTable(ss.getSheetByName("男簡護危疾保保費")), sce_f: parseSceRateTable(ss.getSheetByName("女簡護危疾保保費")) }; const responseData = { status: "success", version: currentVer, config: config, pages: pages, options: options, cases: cases, rules: rules, rates: rates }; if (callback) { return ContentService.createTextOutput(callback + "(" + JSON.stringify(responseData) + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(responseData)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { const errObj = { status: "error", message: err.toString() }; if (e && e.parameter && e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify(errObj) + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(errObj)) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const postData = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const action = postData.action; if (action === "saveAllAdmin") { const pagesData = postData.pages; const optionsData = postData.options; const configData = postData.config; const rulesData = postData.rules; const casesData = postData.cases; // 1. 寫回系統設定 if (configData) { const cfgSheet = ss.getSheetByName("系統設定"); if (cfgSheet) { const vals = cfgSheet.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { const key = vals[i][0]; if (configData[key] !== undefined) { cfgSheet.getRange(i + 1, 3).setValue(configData[key]); } } } } // 2. 寫回頁面文案 (包含 highlightText 及 imageUrl) if (pagesData && Array.isArray(pagesData)) { const pSheet = ss.getSheetByName("頁面文案"); if (pSheet) { pSheet.clearContents(); pSheet.appendRow(["頁面編號", "模式", "標籤名稱", "主題", "副標題說明", "pageId", "pageType", "highlightText", "imageUrl"]); pagesData.forEach(p => { const roleCn = p.role === 'child' ? '兒童' : '成人'; pSheet.appendRow([ p.stepOrder, roleCn, p.tagName, p.title, p.subtitle || "", p.pageId || ("step_ci_" + p.stepOrder), p.pageType || "文案", p.highlightText || "", p.imageUrl || "" ]); }); } } // 3. 寫回選項設定 if (optionsData && Array.isArray(optionsData)) { const optSheet = ss.getSheetByName("選項設定"); if (optSheet) { optSheet.clearContents(); optSheet.appendRow(["頁面所屬", "模式", "選項鍵名", "圖標/Emoji", "選項標題", "預設選取", "排序", "pageId"]); optionsData.forEach((o, idx) => { const pageBelong = o.pageId === 'step_ci_1' ? '01_目標' : (o.pageId === 'step_ci_2' ? '02_資產' : '自訂頁面'); let roleCn = '通用'; if (o.role === 'adult') roleCn = '成人'; else if (o.role === 'child') roleCn = '兒童'; const isDefCn = String(o.isDefaultSelected).toUpperCase() === 'TRUE' ? '是' : '否'; optSheet.appendRow([pageBelong, roleCn, o.optionKey, o.icon, o.title, isDefCn, o.order || (idx + 1), o.pageId]); }); } } // 4. 寫回賠償條款 (Rules) if (rulesData && Array.isArray(rulesData)) { const rSheet = ss.getSheetByName("賠償條款"); if (rSheet) { rSheet.clearContents(); rSheet.appendRow(["diseaseName", "claimCount", "amount", "date", "filesJson", "displayNote"]); const todayStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy/M/d"); rulesData.forEach(r => { let countLabel = `上限:${r.maxCount}次`; if (r.claimKey === 'cancer_b') countLabel = `上限:${r.maxCount}個月`; if (r.claimKey === 'dementia') countLabel = `上限:1次+年金`; let amtLabel = r.isMonthly ? "5%" : (r.claimKey === 'dementia' ? "100%+6%" : "100%"); rSheet.appendRow([ r.name || "", countLabel, amtLabel, todayStr, JSON.stringify(r), r.displayNote || "" ]); }); } } // 5. 寫回賠償案例 (Cases) if (casesData && Array.isArray(casesData)) { const cSheet = ss.getSheetByName("賠償案例"); if (cSheet) { cSheet.clearContents(); cSheet.appendRow(["category", "diseaseName", "filesJson"]); casesData.forEach(c => { cSheet.appendRow([c.category || "", c.diseaseName || "", c.filesJson || "[]"]); }); } } return responseJSON({ status: "success", message: "全部設定已同步寫入 Google Sheet" }); } return responseJSON({ status: "error", message: "未知的操作指令" }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function sheetToObjects(sheet) { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const headers = data[0]; const result = []; for (let i = 1; i < data.length; i++) { let row = data[i]; let obj = { _rowIndex: i + 1 }; for (let h = 0; h < headers.length; h++) { let val = row[h]; if (val instanceof Date) val = Utilities.formatDate(val, "GMT+8", "yyyy/MM/dd"); obj[headers[h]] = val; } result.push(obj); } return result; } function parseRulesSheet(sheet) { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const rules = []; for (let i = 1; i < data.length; i++) { const rawJson = data[i][4]; let parsed = null; try { parsed = JSON.parse(rawJson); } catch(e) {} const customDisplayNote = data[i][5] ? String(data[i][5]).trim() : ''; if (parsed) { if (customDisplayNote) parsed.displayNote = customDisplayNote; rules.push(parsed); } else { rules.push({ claimKey: "rule_" + i, name: data[i][0], title: data[i][0], emoji: "🛡️", maxCount: 3, waitYears: 1, isMonthly: false, color: "#d97706", displayNote: customDisplayNote || data[i][0] }); } } return rules; } function parseOys2RateTable(sheet) { if (!sheet) return {}; const data = sheet.getDataRange().getValues(); const rateMap = {}; for (let i = 4; i < data.length; i++) { const age = data[i][0]; if (age !== "" && !isNaN(age)) { rateMap[Number(age)] = { pay10_non: Number(data[i][1]) || 0, pay10_smk: Number(data[i][2]) || 0, pay18_non: Number(data[i][3]) || 0, pay18_smk: Number(data[i][4]) || 0, pay25_non: Number(data[i][5]) || 0, pay25_smk: Number(data[i][6]) || 0 }; } } return rateMap; } function parseSceRateTable(sheet) { if (!sheet) return {}; const data = sheet.getDataRange().getValues(); const rateMap = {}; for (let i = 3; i < data.length; i++) { const age = data[i][0]; if (age !== "" && !isNaN(age)) { rateMap[Number(age)] = { non_smoker: Number(data[i][1]) || 0, smoker: Number(data[i][2]) || 0 }; } } return rateMap; } function responseJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
